trigger:
  branches:
    include:
      - main  # Trigger pipeline on push to the main branch

pool:
  vmImage: 'ubuntu-latest'  # Use Ubuntu for the build process

steps:
  # 1. Checkout the latest code from Azure Repos
  - task: Checkout@1
    inputs:
      clean: true  # Clean the repository before building

  # 2. Build the Docker image
  - script: |
      echo "Building the Docker image for the front-end app"
      docker build -t my-frontend-app .  # Builds the Docker image
    displayName: 'Build Docker Image'

  # 3. Push the Docker image to Azure Container Registry (ACR)
  - script: |
      echo "Pushing the Docker image to the container registry"
      docker tag my-frontend-app balqisacr.azurecr.io/my-frontend-app:latest  # Tag the image with your ACR URL
      docker push balqisacr.azurecr.io/my-frontend-app:latest  # Push the image to ACR
    displayName: 'Push Docker Image to ACR'
    env:
      ACR_NAME: 'balqisacr'  # Replace with your actual ACR name

  # 4. Deploy to Azure Web App for Containers (optional)
  - script: |
      echo "Deploying the Docker image to Azure"
      az webapp config container set --name my-web-app --resource-group my-resource-group --docker-custom-image-name balqisacr.azurecr.io/my-frontend-app:latest
    displayName: 'Deploy to Azure Web App'
